def rank_1_add2():
    PROTOSSM_W = _weights[0]
    SED_W      = _weights[1]
    EPS        = 1e-5
    PROTO = pd.read_csv(_files_subm[0])
    SED   = pd.read_csv(_files_subm[1]) 
    cols = [c for c in SED.columns if c != "row_id"]
    missing = sorted(set(SED["row_id"]) - set(PROTO["row_id"]))
    if missing:
        raise ValueError(f"row_id mismatch: {len(missing)} rows in ProtoSSM not in SED; first={missing[:3]}")
    PROTO = PROTO.set_index("row_id").loc[SED["row_id"]].reset_index()
    pSED   = np.clip(SED  [cols].to_numpy("float32"), EPS, 1.0 - EPS)
    pPROTO = np.clip(PROTO[cols].to_numpy("float32"), EPS, 1.0 - EPS)
    wSED   = PROTOSSM_W / (PROTOSSM_W + SED_W)
    wPROTO = SED_W      / (PROTOSSM_W + SED_W)
    rSED   = pd.DataFrame(pSED)  .rank(axis=0, pct=True).to_numpy("float32")
    rPROTO = pd.DataFrame(pPROTO).rank(axis=0, pct=True).to_numpy("float32")
    pred = wSED * rSED  +  rPROTO * wPROTO 
    submission = SED.copy()
    submission[cols] = pred.astype("float32") #;submission.to_csv('submission.csv',index=False)
    return submission


def rank_1_add3():
    PROTO_w1    = _weights[0]
    PROTO_w2    = _weights[1]
    SED_w       = _weights[2]
    EPS         = 1e-5
    PROTO_1     = pd.read_csv(_files_subm[0])
    PROTO_2     = pd.read_csv(_files_subm[1])
    SED         = pd.read_csv(_files_subm[2]) 
    cols        = [c for c in SED.columns if c != "row_id"]
    missing1    = sorted(set(SED["row_id"]) - set(PROTO_1["row_id"]))
    missing2    = sorted(set(SED["row_id"]) - set(PROTO_2["row_id"]))
    if missing1 :
        raise ValueError(f"row_id mismatch: {len(missing1)} rows in ProtoSSM not in SED; first={missing1[:3]}")
    if missing2 :
        raise ValueError(f"row_id mismatch: {len(missing2)} rows in ProtoSSM not in SED; first={missing2[:3]}")
    PROTO_1     = PROTO_1.set_index("row_id").loc[SED["row_id"]].reset_index()
    PROTO_2     = PROTO_2.set_index("row_id").loc[SED["row_id"]].reset_index()
    pSED        = np.clip(SED    [cols].to_numpy("float32"), EPS, 1.0 - EPS)
    pPROTO_1    = np.clip(PROTO_1[cols].to_numpy("float32"), EPS, 1.0 - EPS)
    pPROTO_2    = np.clip(PROTO_2[cols].to_numpy("float32"), EPS, 1.0 - EPS)
    wSED        = (PROTO_w1 + PROTO_w2) / ((PROTO_w1+PROTO_w2) + SED_W)
    wPROTO_1    = (SED_W    + PROTO_w2) / ((PROTO_w1+PROTO_w2) + SED_W)
    wPROTO_2    = (SED_W    + PROTO_w1) / ((PROTO_w1+PROTO_w2) + SED_W)
    rSED        = pd.DataFrame(pSED)    .rank(axis=0, pct=True).to_numpy("float32")
    rPROTO_1    = pd.DataFrame(pPROTO_1).rank(axis=0, pct=True).to_numpy("float32")
    rPROTO_2    = pd.DataFrame(pPROTO_2).rank(axis=0, pct=True).to_numpy("float32")
    pred        = wSED * rSED   +   rPROTO_1 * wPROTO_1   +   rPROTO_2 * wPROTO_2 
    subm        = SED.copy()
    subm[cols]  = pred.astype("float32")
    return subm


def _read_submission_checked(path):
    df = pd.read_csv(path)
    assert "row_id" in df.columns, f"row_id column missing in {path}"
    assert df["row_id"].is_unique, f"duplicate row_id values in {path}"
    prob_cols = [c for c in df.columns if c != "row_id"]
    assert prob_cols, f"no probability columns in {path}"
    values = df[prob_cols].to_numpy()
    assert np.isfinite(values).all(), f"NaN/inf values in {path}"
    assert values.min() >= 0.0 and values.max() <= 1.0, f"probabilities outside [0, 1] in {path}"
    return df.set_index("row_id")


def direct_addsafe():
    print(f'Ensemble: {_ensemble_models},   LB: {_lbs},   weights: {_weights}')
    assert len(_files_subm) == len(_weights), "submission file / weight length mismatch"
    weight_sum = float(sum(_weights))
    assert weight_sum > 0, "ensemble weights must sum to a positive value"
    if not np.isclose(weight_sum, 1.0, atol=1e-6):
        print(f"Normalizing ensemble weights from sum={weight_sum:.6f}")
    norm_weights = [float(w) / weight_sum for w in _weights]
    dfs = [_read_submission_checked(path) for path in _files_subm]
    base_idx = dfs[0].index
    base_cols = dfs[0].columns
    for path, df in zip(_files_subm, dfs):
        assert df.columns.equals(base_cols), f"Column mismatch in {path}"
        missing = base_idx.difference(df.index)
        extra = df.index.difference(base_idx)
        assert len(missing) == 0 and len(extra) == 0, (
            f"row_id mismatch in {path}: missing={len(missing)}, extra={len(extra)}"
        )
    out = sum(w * df.loc[base_idx, base_cols] for w, df in zip(norm_weights, dfs))
    values = out.to_numpy()
    assert np.isfinite(values).all(), "NaN/inf in final blend"
    assert values.min() >= 0.0 and values.max() <= 1.0, "final probabilities outside [0, 1]"
    return out

# ---------------------------
# Taxonomy smoothing post-processing
# ---------------------------
def f_TAX_SMOOTHING_POSTPROC(
    f_direct=direct,
    genus_α=0.15,
    class_α=0.05,
):

    # Generate base submission
    submission = f_direct()

    # Import dependencies
    import pandas as pd
    import numpy as np

    # Print initial statistics
    print(
        f"[v221] Pre-tax-smoothing: "
        f"shape={submission.shape}, "
        f"mean={submission.values.mean():.4f}"
    )

    # Define taxonomy paths
    tax_paths = [
        "/kaggle/input/competitions/birdclef-2026/taxonomy.csv",
        "/kaggle/input/birdclef-2026/taxonomy.csv",
    ]

    # Initialize taxonomy dataframe
    taxonomy_df = None

    # Search for taxonomy file
    for taxonomy_path in tax_paths:

        # Check if file exists
        if os.path.exists(taxonomy_path):

            # Load taxonomy
            taxonomy_df = pd.read_csv(taxonomy_path)

            # Stop after first valid file
            break

    # Skip smoothing if taxonomy missing
    if taxonomy_df is None:

        # Print warning
        print("[v221] taxonomy.csv not found — skipping tax smoothing")

    else:

        # Initialize mappings
        species_to_genus = {}
        species_to_class = {}

        # Build taxonomy mappings
        for _, row in taxonomy_df.iterrows():

            # Extract taxonomy fields
            label = str(row["primary_label"])
            scientific_name = str(row.get("scientific_name", ""))
            class_name = str(row.get("class_name", ""))

            # Extract genus from scientific name
            genus = (
                scientific_name.split(" ")[0]
                if " " in scientific_name
                else scientific_name
            )

            # Store mappings
            species_to_genus[label] = genus
            species_to_class[label] = class_name

        # Get submission columns
        columns = list(submission.columns)

        # Initialize grouping dictionaries
        genus_groups = {}
        class_groups = {}

        # Build genus/class groups
        for column in columns:

            # Resolve taxonomy
            genus = species_to_genus.get(column, column)
            class_name = species_to_class.get(column, "")

            # Add genus member
            genus_groups.setdefault(genus, []).append(column)

            # Add class member
            if class_name:
                class_groups.setdefault(class_name, []).append(column)

        # Keep only multi-member genus groups
        multi_genus = {
            genus: members
            for genus, members in genus_groups.items()
            if len(members) > 1
        }

        # Keep only multi-member class groups
        multi_class = {
            class_name: members
            for class_name, members in class_groups.items()
            if len(members) > 1
        }

        # Print group statistics
        print(
            f"[v221] multi-genus groups: {len(multi_genus)}, "
            f"multi-class groups: {len(multi_class)}"
        )

        # Store smoothing coefficients
        alpha_genus = genus_α
        alpha_class = class_α

        # Copy prediction probabilities
        probs = submission.values.astype(np.float32).copy()

        # ---------------------------
        # Genus-level smoothing
        # ---------------------------
        for genus, members in multi_genus.items():

            # Resolve column indices
            indices = [
                columns.index(member)
                for member in members
            ]

            # Compute genus mean
            genus_mean = probs[:, indices].mean(
                axis=1,
                keepdims=True,
            )

            # Apply smoothing
            probs[:, indices] = (
                (1 - alpha_genus) * probs[:, indices]
                + alpha_genus * genus_mean
            )

        # ---------------------------
        # Class-level smoothing
        # ---------------------------
        for class_name, members in multi_class.items():

            # Resolve column indices
            indices = [
                columns.index(member)
                for member in members
            ]

            # Compute class mean
            class_mean = probs[:, indices].mean(
                axis=1,
                keepdims=True,
            )

            # Apply smoothing
            probs[:, indices] = (
                (1 - alpha_class) * probs[:, indices]
                + alpha_class * class_mean
            )

        # Rebuild submission dataframe
        submission = pd.DataFrame(
            probs,
            index=submission.index,
            columns=submission.columns,
        )

        # Print final statistics
        print(
            f"[v221] Post-tax-smoothing: "
            f"mean={submission.values.mean():.4f}"
        )

    # Return final submission
    return submission
